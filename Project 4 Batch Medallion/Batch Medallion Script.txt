#--BUILD GOLD LAYER NOTE: THIS FIRST VIEW IS NOT GOLD; IT is an intermediate between silver and gold
%sql

USE CATALOG AdjustedCatalogName;
USE SCHEMA TiogamDBProjects;

CREATE OR REPLACE TEMP VIEW Temp_Payouts AS
SELECT
    AccountName,
    SUM(Amount) As TotalOut
FROM SilverDemo
WHERE AccountName = 'PayOutsAccount'
GROUP BY AccountName;

%sql
SELECT * FROM Temp_Payouts;

%sql
USE CATALOG AdjustedCatalogName;
USE SCHEMA TiogamDBProjects;

CREATE OR REPLACE TEMP VIEW Temp_Income AS
SELECT 
    AccountName,
    SUM(Amount) AS TotalIn
FROM SilverDemo
WHERE AccountName = 'IncomeAccount'
GROUP BY AccountName;
